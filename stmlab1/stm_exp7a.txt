import java.util.Random;

public class DynamicDataTest {

    public static double calculateDiscount(
            double price, String category) {

        switch (category) {
            case "electronics":
                return price * 0.9;

            case "clothing":
                return price * 0.8;

            default:
                return price;
        }
    }

    public static void main(String[] args) {

        String[] categories =
            {"electronics", "clothing", "grocery"};

        Random rand = new Random();

        System.out.println(
            "---- Dynamic Test Data Submission ----");

        for (int i = 1; i <= 5; i++) {

            int price = 100 + rand.nextInt(901);

            String category =
                categories[rand.nextInt(categories.length)];

            double result =
                calculateDiscount(price, category);

            System.out.printf(
                "Test %d: price=%d, category=%s -> discounted=%.2f%n",
                i, price, category, result);
        }
    }
}
