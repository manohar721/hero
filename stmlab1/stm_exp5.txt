import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.Statement;

public class NewTest {

    public static void main(String[] args) {
        try {
            // 1. Establish connection to local embedded H2 database
            Connection con = DriverManager.getConnection(
                    "jdbc:h2:./StudentDB", "sa", "");

            Statement stmt = con.createStatement();

            // 2. Create schema
            stmt.execute(
                    "CREATE TABLE IF NOT EXISTS STUDENT (" +
                    "ID INT PRIMARY KEY, " +
                    "NAME VARCHAR(50), " +
                    "EMAIL VARCHAR(100))");

            // 3. Clear previous test artifacts
            stmt.execute("DELETE FROM STUDENT");

            // 4. Insert test record
            stmt.executeUpdate(
                    "INSERT INTO STUDENT VALUES " +
                    "(1, 'Ajay', 'ajay@gmail.com')");

            // 5. Query the database
            ResultSet rs = stmt.executeQuery(
                    "SELECT * FROM STUDENT WHERE ID = 1");

            // 6. Database Checkpoint Verification
            if (rs.next()) {
                System.out.println("Record Exists");
                System.out.println("ID    : " + rs.getInt("ID"));
                System.out.println("Name  : " + rs.getString("NAME"));
                System.out.println("Email : " + rs.getString("EMAIL"));
                System.out.println("TEST PASSED");
            } else {
                System.out.println("Record Not Found");
                System.out.println("TEST FAILED");
            }

            // Clean up resources
            rs.close();
            stmt.close();
            con.close();

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
